set(PERF_VARIANTS libm h32 h64 p8 p12)
set(PERF_NAMES libm hermite32 hermite64 polynomial8 polynomial12)
foreach(index RANGE 0 4)
    list(GET PERF_VARIANTS ${index} variant)
    list(GET PERF_NAMES ${index} label)
    add_library(germanium_perf_${variant} STATIC ../../src/parable_germanium.cpp perf_bridge.cpp)
    target_include_directories(germanium_perf_${variant} PRIVATE ../../include ../../src)
    target_compile_definitions(germanium_perf_${variant} PRIVATE parable=parable_gp_${variant} PARABLE_GERMANIUM_EVALUATOR=${index} PERF_FACTORY=perf_${variant} PERF_VARIANT_NAME="${label}")
    target_compile_options(germanium_perf_${variant} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -fno-finite-math-only)
endforeach()
