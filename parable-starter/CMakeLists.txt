cmake_minimum_required(VERSION 3.15)
project(parable_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(parable_core_test tests/test_core.cpp)
target_include_directories(parable_core_test PRIVATE include)
target_compile_options(parable_core_test PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME core_validation COMMAND parable_core_test)
